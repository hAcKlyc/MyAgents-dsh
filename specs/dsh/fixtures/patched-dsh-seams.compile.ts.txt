import type { Context } from "@deepseek-ai/cordis";
import type {
  Agent,
  AgentPublicationGuard,
  PreparedAssistantCommit,
  PreparedAssistantToolCall,
} from "@deepseek-ai/dsh-agent";
import type { MessageId } from "@deepseek-ai/dsh-llm";
import type { SessionPublicationGuard } from "@deepseek-ai/dsh-session";
import type { SessionPersistence } from "@deepseek-ai/dsh-session-persistence";
import type { ContinuableSubagentDescriptorData } from "@deepseek-ai/dsh-subagent";
import {
  SUMMARY_SAFETY_MARGIN_TOKENS,
  SummaryCapacityError,
  SummaryValidationError,
  type CompactionTelemetry,
} from "@deepseek-ai/dsh-compaction-basic";
import type { TokenRequestEstimateInput } from "@deepseek-ai/dsh-token-meter";

export interface PatchedDshSeamTypes {
  readonly commit: PreparedAssistantCommit;
  readonly toolCall: PreparedAssistantToolCall;
}

export const compilePatchedDshSeams = (
  ctx: Context,
  agent: Agent,
  messageId: MessageId,
  persistence: SessionPersistence,
): SessionPersistence => {
  const agentGuard: AgentPublicationGuard = () => undefined;
  const sessionGuard: SessionPublicationGuard = () => undefined;
  const disposeAgentGuard = ctx.agents.setPublicationGuard(agentGuard);
  const disposeSessionGuard = ctx.sessions.setPublicationGuard(sessionGuard);
  disposeSessionGuard();
  disposeAgentGuard();
  const woke: boolean | undefined = agent.wakePending?.(messageId);
  void woke;
  const waitForChildren = ctx.subagents.waitForChildren;
  const drainDescendants = ctx.subagents.drainDescendants;
  const startActivation = ctx.subagents.startActivation;
  const settlementDelivery: ContinuableSubagentDescriptorData["settlementDelivery"] = "external";
  const literalPersona: ContinuableSubagentDescriptorData["personaInterpolate"] = false;
  void waitForChildren;
  void drainDescendants;
  void startActivation;
  void literalPersona;
  const requestEstimate: TokenRequestEstimateInput = { messages: [], route: { provider: "synthetic", model: "summary" } };
  const estimatedTokens: number = ctx.tokenMeter.estimateRequest(requestEstimate);
  const safetyMargin: 1024 = SUMMARY_SAFETY_MARGIN_TOKENS;
  const capacityError = new SummaryCapacityError("compile fixture");
  const validationError = new SummaryValidationError("compile fixture");
  const disposeTelemetry = ctx.on("compaction/telemetry", (event) => {
    const telemetry: Readonly<CompactionTelemetry> = event;
    void telemetry;
  });
  disposeTelemetry();
  void settlementDelivery;
  void estimatedTokens;
  void safetyMargin;
  void capacityError;
  void validationError;
  ctx.on("agent/pre-assistant-commit", async ({ commit, signal }, next) => {
    const prepared: Readonly<PreparedAssistantCommit> = commit;
    signal.throwIfAborted();
    void prepared;
    return next();
  });
  return persistence;
};

import { withStoredEventValidation } from "@deepseek-ai/dsh-session-persistence-jsonl";

export const trustedJsonlPlugin = withStoredEventValidation((event) => {
  void event.data;
  return false;
});
